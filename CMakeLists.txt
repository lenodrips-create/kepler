cmake_minimum_required(VERSION 3.16)

project(Kepler VERSION 1.1.0 DESCRIPTION "Kepler - a web browser from another orbit" LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network WebEngineCore WebEngineWidgets Svg)

set(KEPLER_SOURCES
    src/main.cpp
    src/Browser.cpp            src/Browser.h
    src/AiPanel.cpp            src/AiPanel.h
    src/MacWindow.h
    src/BrowserWindow.cpp      src/BrowserWindow.h
    src/WebView.cpp            src/WebView.h
    src/UrlBar.cpp             src/UrlBar.h
    src/Favicons.cpp           src/Favicons.h
    src/FindBar.cpp            src/FindBar.h
    src/DownloadsPanel.cpp     src/DownloadsPanel.h
    src/Storage.cpp            src/Storage.h
    src/Shields.cpp            src/Shields.h
    src/SchemeHandler.cpp      src/SchemeHandler.h
    src/Theme.cpp              src/Theme.h
    resources/kepler.qrc
)

if(WIN32)
    list(APPEND KEPLER_SOURCES resources/kepler.rc)
endif()

# Mac: native tweaks that blend the title bar into Kepler's top bar
if(APPLE)
    enable_language(OBJCXX)
    list(APPEND KEPLER_SOURCES src/MacWindow.mm)
endif()

# Mac app icon (resources/kepler.icns is generated by the GitHub build)
if(APPLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/resources/kepler.icns")
    set(KEPLER_ICNS "${CMAKE_CURRENT_SOURCE_DIR}/resources/kepler.icns")
    set_source_files_properties(${KEPLER_ICNS} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")
    list(APPEND KEPLER_SOURCES ${KEPLER_ICNS})
endif()

qt_add_executable(Kepler WIN32 MACOSX_BUNDLE ${KEPLER_SOURCES})

target_compile_definitions(Kepler PRIVATE KEPLER_VERSION="${PROJECT_VERSION}")
target_link_libraries(Kepler PRIVATE
    Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network
    Qt6::WebEngineCore Qt6::WebEngineWidgets Qt6::Svg)
if(APPLE)
    target_link_libraries(Kepler PRIVATE "-framework AppKit")
endif()

set_target_properties(Kepler PROPERTIES
    MACOSX_BUNDLE_BUNDLE_NAME "Kepler"
    MACOSX_BUNDLE_GUI_IDENTIFIER "app.kepler.browser"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_ICON_FILE "kepler.icns")

install(TARGETS Kepler BUNDLE DESTINATION . RUNTIME DESTINATION bin)

# Linux: app menu entry + icon, and a .deb package (cpack -G DEB)
if(UNIX AND NOT APPLE)
    install(FILES packaging/linux/kepler.desktop DESTINATION share/applications)
    install(FILES resources/logo-icon.png DESTINATION share/icons/hicolor/512x512/apps RENAME kepler.png)

    set(CPACK_PACKAGE_NAME "kepler")
    set(CPACK_PACKAGE_VENDOR "Kepler")
    set(CPACK_PACKAGE_CONTACT "Kepler <noreply@kepler.app>")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "${PROJECT_DESCRIPTION}")
    set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
    set(CPACK_DEBIAN_PACKAGE_SECTION "web")
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    # Plugins Qt loads at runtime (SVG icons, the web engine helper process)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6svg6, libqt6webenginecore6-bin")
    include(CPack)
endif()
